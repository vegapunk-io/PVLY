Examples

# synchronous scan of a file
curl -s -F "file=@photo.jpg" -F "wait=true" {{ request.base_url }}v1/scans | jq .conclusion

# async: create, then poll
curl -s -F "file=@clip.mp4" {{ request.base_url }}v1/scans
curl -s {{ request.base_url }}v1/scans/<id>

# disclosure checklist
curl -s -X POST {{ request.base_url }}v1/disclosure/check -H "content-type: application/json" \
  -d '{"content_type":"video","is_ai_generated":true,"depicts_real_person":true,"has_c2pa":false}'
